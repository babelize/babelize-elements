## Usage

```tsx
"use client";

import { useState } from "react";
import { PhoneInput } from "@/components/ui/phone-input";

export function PhoneField() {
  const [phone, setPhone] = useState("");

  return (
    <PhoneInput
      defaultCountry="US"
      placeholder="Enter phone number"
      value={phone} // omit for uncontrolled
      onValueChange={(value, country) => {
        setPhone(value);
        console.log(country.dialCode);
      }}
    />
  );
}
```

## Props

| Prop                | Type                                        | Required | Description                                                                                         |
| ------------------- | ------------------------------------------- | -------- | --------------------------------------------------------------------------------------------------- |
| `value`             | `string`                                    | No       | Phone number — pass this to control the component                                                   |
| `defaultValue`      | `string`                                    | No       | Initial phone number when uncontrolled                                                              |
| `onValueChange`     | `(phone: string, country: Country) => void` | No       | Callback when phone number or country changes                                                       |
| `defaultCountry`    | `string`                                    | No       | Default country code (ISO 3166-1 alpha-2, e.g. `"US"`, `"GB"`, `"IN"`)                              |
| `countries`         | `Country[]`                                 | No       | Countries offered, in display order (default: `COUNTRIES`)                                          |
| `placeholder`       | `string`                                    | No       | Placeholder text                                                                                    |
| `disabled`          | `boolean`                                   | No       | Disable the input                                                                                   |
| `showFlags`         | `boolean`                                   | No       | Show country flag next to dial code (default: `true`)                                               |
| `size`              | `"sm" \| "default" \| "lg"`                 | No       | Field height (default: `"default"`)                                                                 |
| `align`             | `"start" \| "end"`                          | No       | Edge of the field the menu lines up with (default: `"start"`)                                       |
| `side`              | `"top" \| "bottom"`                         | No       | Preferred menu side; flips when there is no room (default: `"bottom"`)                              |
| `searchPlaceholder` | `string`                                    | No       | Placeholder for the country search (default: `"Search countries..."`)                               |
| `emptyMessage`      | `ReactNode`                                 | No       | Shown when the search matches nothing (default: `"No countries found."`)                            |
| `labels`            | `Partial<PhoneInputLabels>`                 | No       | Override the built-in English UI text (see below)                                                   |
| `displayLocale`     | `string`                                    | No       | Locale for country names, e.g. `"fr"` (default: English names)                                      |
| `maxHeight`         | `number \| string`                          | No       | Max height of the scrolling country list (default: `"15rem"`)                                       |
| `label`             | `string`                                    | No       | Accessible label for the input                                                                      |
| `className`         | `string`                                    | No       | Classes for the root wrapper                                                                        |
| `classNames`        | `PhoneInputClassNames`                      | No       | Per-part classes: `root`, `field`, `trigger`, `input`, `content`, `search`, `list`, `item`, `empty` |

Any other prop — `name`, `required`, `autoComplete`, `aria-invalid`, `data-*` — is
spread onto the underlying `<input>`, and the ref is forwarded to it, so the
component works inside a plain HTML form. Setting `aria-invalid` turns the whole
field red, matching shadcn's `Input`.

### Country Object

| Property   | Type     | Description                                           |
| ---------- | -------- | ----------------------------------------------------- |
| `code`     | `string` | ISO 3166-1 alpha-2 code (e.g. `"US"`, `"GB"`, `"IN"`) |
| `dialCode` | `string` | Dial code (e.g. `"+1"`, `"+44"`, `"+91"`)             |
| `name`     | `string` | Country name                                          |
| `flag`     | `string` | Flag emoji                                            |

## Examples

### With Babelize SDK

Pair Phone Input with the [Babelize SDK](https://docs.babelize.co/sdk) for locale-aware validation:

```tsx
"use client";

import { useState } from "react";
import { PhoneInput } from "@/components/ui/phone-input";
import { useBabelize } from "@babelize/sdk/react";

export function ContactPhone() {
  const { locale } = useBabelize();
  const [phone, setPhone] = useState("");

  return (
    <PhoneInput
      value={phone}
      onValueChange={(value, country) => {
        setPhone(value);
        // Validate against locale-specific rules
      }}
      defaultCountry={locale === "en" ? "US" : locale === "de" ? "DE" : "GB"}
      placeholder="Enter phone number"
    />
  );
}
```

### With Form Validation

```tsx
"use client";

import { useState } from "react";
import { PhoneInput } from "@/components/ui/phone-input";

export default function CheckoutForm() {
  const [phone, setPhone] = useState("");
  const [error, setError] = useState("");

  const validate = (value: string) => {
    if (value.length < 10) {
      setError("Phone number is too short");
    } else {
      setError("");
    }
  };

  return (
    <div>
      <PhoneInput
        value={phone}
        onValueChange={(value) => {
          setPhone(value);
          validate(value);
        }}
        defaultCountry="US"
        placeholder="Enter phone number"
        aria-invalid={error ? true : undefined}
      />
      {error && <p className="mt-1 text-sm text-destructive">{error}</p>}
    </div>
  );
}
```

### With Country Filter

```tsx
"use client";

import { COUNTRIES, PhoneInput } from "@/components/ui/phone-input";

// Only the markets you ship to, in the order you want them listed.
const SUPPORTED = ["US", "GB", "CA", "AU"].map((code) =>
  COUNTRIES.find((country) => country.code === code)!,
);

export default function App() {
  return (
    <PhoneInput
      countries={SUPPORTED}
      defaultCountry="US"
      onValueChange={(phone, country) => console.log(phone, country)}
    />
  );
}
```

### Translated UI text

Pass `labels` to translate the country menu's text, and `displayLocale` to show
country names in another language. Names come from the browser's
[`Intl.DisplayNames`](https://developer.mozilla.org/en-US/docs/Web/JavaScript/Reference/Global_Objects/Intl/DisplayNames),
so their exact wording can vary slightly between browsers. Searching matches the
translated name, while `onValueChange` still reports the country's English `name`.
The `searchPlaceholder` and `emptyMessage` props still work, and win over `labels`
when both are set.

```tsx
"use client";

import { PhoneInput } from "@/components/ui/phone-input";

export default function App() {
  return (
    <PhoneInput
      defaultCountry="FR"
      displayLocale="fr"
      label="Numéro de téléphone"
      placeholder="Numéro de téléphone"
      labels={{
        trigger: "Choisir un pays",
        searchPlaceholder: "Rechercher un pays…",
        searchLabel: "Rechercher un pays",
        clearSearch: "Effacer la recherche",
        emptyMessage: "Aucun pays trouvé.",
        listbox: "Choisir un pays",
      }}
    />
  );
}
```

| Key                 | Default                 | Used for                      |
| ------------------- | ----------------------- | ----------------------------- |
| `trigger`           | `"Select country"`      | Country trigger label         |
| `searchPlaceholder` | `"Search countries..."` | Search field placeholder      |
| `searchLabel`       | `"Search countries"`    | Search field accessible label |
| `clearSearch`       | `"Clear search"`        | Clear button accessible label |
| `emptyMessage`      | `"No countries found."` | Shown when nothing matches    |
| `listbox`           | `"Select country"`      | Country list accessible label |

## Keyboard support

The country menu follows the WAI-ARIA combobox pattern: focus stays in the search
field, which is exposed as a `role="combobox"`, and the highlighted country is
announced through `aria-activedescendant`.

| Key                              | Action                                                                |
| -------------------------------- | --------------------------------------------------------------------- |
| `Enter`, `Space`, or click       | Open the country menu from the flag trigger                           |
| `ArrowDown` / `ArrowUp` (closed) | Open with the first / last country highlighted                        |
| `ArrowDown` / `ArrowUp`          | Move the highlight, wrapping at both ends                             |
| `Home` / `End`                   | Highlight the first / last country                                    |
| `Enter`                          | Select the highlighted country; focus moves to the number field       |
| `Escape`                         | Close the menu and return focus to the flag trigger                   |
| Typing                           | Filter by name, code, or dial code; the highlight restarts at the top |

The highlight stays on the row where it appeared when the menu opened — the
list scrolls beneath it, like a native menu — and the page itself never
scrolls. Pass `maxHeight` to change how tall the country list gets before it
scrolls.

## Features

- **50 countries** — pre-loaded with dial codes, flags, and names
- **Dropdown search** — filter countries by name or dial code
- **Flag emojis** — auto-detected from country code, with `showFlags` toggle
- **Keyboard accessible** — arrow keys, Home / End, Enter, and Escape navigation
- **ARIA labels** — `role="combobox"` search field with `aria-activedescendant`, plus `aria-expanded`, `aria-haspopup`, and `aria-selected`
- **Themeable** — styled with shadcn/ui tokens (`border-input`, `bg-popover`, `ring-ring`), so it follows your theme and dark mode
- **Animated menu** — fades in and out with `motion`, honouring reduced-motion settings, and flips upward when there is no room below
- **Controlled/uncontrolled** — works with or without `value` prop
