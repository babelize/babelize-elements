## Usage

```tsx
import { LanguageSwitcher } from "@babelize/elements";

export default function App() {
  const [locale, setLocale] = useState("en");

  return (
    <LanguageSwitcher
      locale={locale}
      locales={[
        { code: "en", label: "English", flag: "🇺🇸" },
        { code: "fr", label: "Français", flag: "🇫🇷" },
        { code: "es", label: "Español", flag: "🇪🇸" },
        { code: "de", label: "Deutsch", flag: "🇩🇪" },
        { code: "ja", label: "日本語", flag: "🇯🇵" },
        { code: "ar", label: "العربية", flag: "🇸🇦", rtl: true },
      ]}
      onLocaleChange={(code) => {
        // next-intl
        router.push(`/${code}`);
        // react-i18next
        i18n.changeLanguage(code);
      }}
    />
  );
}
```

## Props

| Prop | Type | Required | Description |
|------|------|----------|-------------|
| `locale` | `string` | Yes | Current active locale code |
| `locales` | `Locale[]` | Yes | Array of available locales |
| `onLocaleChange` | `(code: string) => void` | Yes | Callback when locale changes |
| `className` | `string` | No | Additional CSS classes |

### Locale Object

| Property | Type | Description |
|----------|------|-------------|
| `code` | `string` | ISO 639-1 / BCP 47 code (e.g. `"en"`, `"fr-FR"`, `"ar-SA"`) |
| `label` | `string` | Display name in its native language |
| `flag` | `string` | Optional flag emoji |
| `rtl` | `boolean` | Set `true` for RTL locales (auto-detected from code if omitted) |

## Examples

### With next-intl

```tsx
"use client";

import { useRouter } from "next/navigation";
import { LanguageSwitcher } from "@babelize/elements";

export function LocaleSwitcher() {
  const router = useRouter();
  const [locale, setLocale] = useState("en");

  return (
    <LanguageSwitcher
      locale={locale}
      locales={[
        { code: "en", label: "English", flag: "🇺🇸" },
        { code: "fr", label: "Français", flag: "🇫🇷" },
      ]}
      onLocaleChange={(code) => {
        setLocale(code);
        router.push(`/${code}`);
      }}
    />
  );
}
```

### With react-i18next

```tsx
import { useTranslation } from "react-i18next";
import { LanguageSwitcher } from "@babelize/elements";

export function LocaleSwitcher() {
  const { i18n } = useTranslation();

  return (
    <LanguageSwitcher
      locale={i18n.language}
      locales={[
        { code: "en", label: "English", flag: "🇺🇸" },
        { code: "fr", label: "Français", flag: "🇫🇷" },
      ]}
      onLocaleChange={(code) => i18n.changeLanguage(code)}
    />
  );
}
```

## Features

- **Dropdown with search** — filter locales by name or code
- **RTL-aware** — auto-detects RTL locales (ar, he, fa, ur) and adjusts layout
- **Keyboard accessible** — arrow keys, Enter, Escape, Tab navigation
- **ARIA labels** — proper `aria-expanded`, `aria-haspopup`, `aria-selected`
- **Dark mode** — built with Tailwind CSS, dark-mode compatible
- **Library agnostic** — works with any i18n setup via simple callback
